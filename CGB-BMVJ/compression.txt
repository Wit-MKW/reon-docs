20000 = InstallMinigame(a: u8 firstBank, de: *laterDestinationCallback(a: laterDestinationBank), hl: u8 *source) {
	SVBK = 1
	{D000..DC0F} = 0
	emergencySP = sp
	{0800} = sourceBank = firstBank
	loadedKB = 0
	loadedBytesOfKB = 0
	totalKB = totalBanks << 3 | totalBanks >> 5
	hl = zeroProgressString
	de = progressString
	do {
		a = {hl++}
		{de++} = a
	} while (a != 0)
	h, l = APIByteDivide(totalKB, 10)
	progressString[15] += l
	progressString[16] += h
	{DBF7} = {C1C2}
	APITextClear()
	APIDrawString(progressString, 0, 0)
	crc = 0
	crc = (crc >> 8) ^ crcTable[crc.low ^ 0x01]
	compressedRemain = 0
	uncompressedRemain = 0
	{0000} = 0xA
	hl = ParseHeader(source)
	{0000} = 0
	laterDestinationBank = 4
	CopyMinigame(hl, 0xD000)
	SVBK = 0
}

200FB = ParseHeader(hl: u8 *source) hl: u8* {
	c = {hl++}
	hl += c
	{D00D} = {hl++}
	headerCompressed = {hl++}
	compressedRemain = compressedSize = {hl++} | {hl++} << 8
	uncompressedRemain = uncompressedSize = {hl++} | {hl++} << 8
	{D00B} = {D005} = {hl++} | {hl++} << 8
	return hl
}

20133 = CopyMinigame(hl: u8 *source, de: u8 *laterDestination) a: u8 {
	crc = 0
	compressed = headerCompressed
	if (compressed != NO) {
		if (compressed != YES) return 1
		InitDecompression()
	}
	while (uncompressedRemain != 0) {
		if (uncompressedRemain > 512) {
			bc = 512
		} else {
			bc = uncompressedRemain
		}
		if (compressed != NO) {
			DecompressBlock(bc, destination)
		} else {
			CopyUncompressedBlock(bc)
		}
		UpdateCRC(bc)
		uncompressedRemain -= bc
		if (uncompressedRemain <= 0) {
			if (laterDestination = 0xD000) break
			laterDestinationCallback(laterDestinationBank)
			break
		}
	}
}

201D9 = UpdateCRC(bc: u16 blockSize) {
	UpdateProgress(blockSize)
	hl = destination
	repeat (blockSize or 65536) {
		crc = (crc >> 8) ^ crcTable[crc.low ^ {hl++}]
	}
}

2024F = GetBits(a: u8 bits) bc: u16 {
	bc = inputBits >> (16 - bits)
	DropBits(bits)
	return bc
}

20273 = FinishTree(a: u8 fastBits, bc: u16 symbolCount, de: u8 *bitCounts, hl: u16 *fastCode) {
	bitCountCounts[0..16] = 0
	repeat (symbolCount) {
		bitCountCounts[{de++}].low++
	}
	hl = &nextCodeEntry[1]
	de = &bitCountCounts[1]
	nextCodeEntry[1] = 0
	for (i: 1..16) {
		bc = {hl++} + ({de++} << (16 - i))
		{hl} = bc
	}
	slowBits = 16 - fastBits
	hl = &nextCodeEntry[1]
	de = &codeEntrySize[1]
	for (i: 1..fastBits) {
		{hl++} >>= slowBits
		{de++} = 1 << (fastBits - i)
	}
	for (i: (fastBits+1)..16) {
		{de++} = 1 << (16 - i)
	}
	bc = nextCodeEntry[1 + fastBits] >> slowBits
	if (bc != 0) {
		de = 1 << fastBits
		hl = &fastCode[bc]
		if (hl >= 0xE000) {
			hl -= 0x1000
			SVBK = 3
		} else {
			SVBK = firstWramBank
		}
		while (bc != de) {
			{hl++} = 0
			bc++
			if (0xE000 <= hl <= 0xE0FF) {
				hl -= 0x1000
				SVBK = 3
			}
		}
		SVBK = 1
	}
	nextSymbol = symbolCount
	bc = 1 << (15 - fastBits)
	for (i: 0..(symbolCount-1)) {
		hl = &bitCounts[i]
		if ({hl} != 0) {
			bitCount = {hl}
			codeEntryEnd = nextCodeEntry[bitCount] + codeEntrySize[bitCount]
			if (bitCount <= fastBits) {
				de = nextCodeEntry[bitCount]
				hl = &fastCode[de]
				if (hl >= 0xE000) {
					hl -= 0x1000
					SVBK = 3
				} else {
					SVBK = firstWramBank
				}
				while (de != codeEntryEnd) {
					{hl++} = i
					de++
					if (0xE000 <= hl <= 0xE0FF) {
						hl -= 0x1000
						SVBK = 3
					}
				}
				SVBK = 1
			} else {
				slowCodeEntry = nextCodeEntry[bitCount]
				hl = &fastCode[slowCodeEntry >> slowBits]
				if (hl >= 0xE000) {
					hl -= 0x1000
					SVBK = 3
				} else {
					SVBK = firstWramBank
				}
				for (j: (bitCount-fastBits)..0) {
					slowCodeSymbol = {hl}
					if (slowCodeSymbol = 0) {
						{hl} = slowCodeSymbol = nextSymbol
						SVBK = 1
						slowCodeBranch1[nextSymbol] = 0
						slowCodeBranch0[nextSymbol] = 0
						nextSymbol++
					}
					SVBK = 1
					if ((slowCodeEntry & bc) != 0) {
						hl = slowCodeBranch1[slowCodeSymbol]
					} else {
						hl = slowCodeBranch0[slowCodeSymbol]
					}
					slowCodeEntry <<= 1
				}
				{hl++} = i
				SVBK = 1
			}
			nextCodeEntry[bitCount] = codeEntryEnd
		}
	}
}

205E8 = MakeAuxTree(a: u8 maxSymbols, b: u8 bits, c: u8 skipSymbol) {
	bc = GetBits(bits)
	if (bc = 0) {
		// this code path does not work in any intended or useful way.
		bc = GetBits(bits)
		auxFastCode[0..((bc >> 8 or 256) - 1)] = (u16)e // the MSBs of inputByte, in their position as the LSBs of inputBits.
		auxBitCounts[0..(maxSymbols-1)] = 0
	} else {
		i = 0
		while (i < bc) {
			bc' = inputBits >> 13
			if (bc' < 256) d = (byte)bc'
			if (bc' = 7) {
				bc' = 0x1000
				while ((inputBits & bc') != 0) {
					bc' >>= 1
					d++
				}
			}
			DropBits(max(d, 6) - 3)
			hl = &auxBitCounts[i++]
			{hl} = d
			if (i % 256 = skipSymbol) {
				bc' = GetBits(2)
				hl++
				while (--bc' >= 0) {
					{hl++} = 0
					i++
				}
			}
		}
		hl = &auxBitCounts[i]
		while (i < (u16)maxSymbols) {
			{hl++} = 0
			i++
		}
		firstWramBank = 1
		FinishTree(8, (u16)maxSymbols, auxBitCounts, auxFastCode)
	}
}

206FD = MakeMainTree() {
	bc = GetBits(9)
	if (bc = 0) {
		bc = GetBits(9)
		for (SVBK: 2..3) {
			mainFastCode[0..2047] = bc
		}
		SVBK = 1
		mainBitCounts[0..261] = 0
	} else {
		symbol = 0
		while (symbol < bc) {
			de = auxFastCode[inputBits >> 8]
			bc' = 0x80
			while (de >= 19) {
				if ((inputBits & bc') != 0) {
					de = slowCodeBranch1[de]
				} else {
					de = slowCodeBranch0[de]
				}
				bc' >>= 1
			}
			DropBits(auxBitCounts[de])
			if (de < 3) {
				switch (de) {
				case 0:
					bc' = 1
					break
				case 1:
					bc' = GetBits(4) + 3
					break
				case 2:
					bc' = GetBits(9) + 20
				}
				hl = &mainBitCounts[symbol]
				symbol += bc
				if (bc != 0) {
					hl[0..(bc-1)] = 0
				}
			} else {
				hl = &mainBitCounts[symbol]
				symbol++
				{hl} = (byte)de - 2
			}
		}
		de = symbol
		hl = &mainBitCounts[de]
		while (de < 262) {
			{hl++} = 0
			de++
		}
		firstWramBank = 2
		FinishTree(12, 262, mainBitCounts, mainFastCode)
	}
}

20869 = GetMainSymbol() hl: u16 {
	if (chunkRemain = 0) {
		chunkRemain = GetBits(16)
		MakeAuxTree(19, 5, 3)
		MakeMainTree()
		MakeAuxTree(10, 4, -1)
	}
	chunkRemain--
	bc = (inputBits >> 4) * 2
	if (bc < 4096) {
		SVBK = 2
	} else {
		bc -= 4096
		SVBK = 3
	}
	de = mainFastCode[bc/2]
	SVBK = 1
	bc = 8
	while (de >= 262) {
		if ((inputBits & bc) != 0) {
			de = slowCodeBranch1[de]
		} else {
			de = slowCodeBranch0[de]
		}
		bc >>= 1
	}
	DropBits(mainBitCounts[de])
	return de
}

20922 = GetAuxSymbol10() hl: u16 {
	de = auxFastCode[inputBits >> 8]
	bc = 0x80
	while (de >= 10) {
		if ((inputBits & bc) != 0) {
			de = slowCodeBranch1[de]
		} else {
			de = slowCodeBranch0[de]
		}
		bc >>= 1
	}
	DropBits(auxBitCounts[de])
	if (de != 0) {
		hl = (1 << (de - 1)) + GetBits(de - 1)
	} else {
		hl = 0
	}
	return hl
}

2098B = DecompressBlock(bc: u16 blockSize, de: u8 *destination) {
	hl = de + repeatOffset
	while (--repeatLength >= 0) {
		{de++} = {hl++}
		repeatOffset = (repeatOffset + 1) & 0x1FF
		if (repeatOffset = 0) hl = destination
		if (--bc = 0) return
	}
	until-return {
		hl = GetMainSymbol()
		if (hl < 256) {
			{de++} = (byte)hl
			if (--bc = 0) return
		} else {
			repeatLength = hl - 253
			hl = GetAuxSymbol10() + 1
			repeatOffset = (blockSize - bc - hl) & 0x1FF
			hl = destination + repeatOffset
			while (--repeatLength >= 0) {
				{de++} = {hl++}
				repeatOffset = (repeatOffset + 1) & 0x1FF
				if (repeatOffset = 0) hl = destination
				if (--bc = 0) return
			}
		}
	}
}

20AA7 = DropBits(a: bits) {
	b = bits
	inputBits <<= b
	while (b > bitsLeft.low) {
		b -= bitsLeft.low
		inputBits |= (inputByte << b)
		if ({header1} != 0) {
			{header1}--
			inputByte = (u16)NextByte()
		} else {
			inputByte = 0
		}
		bitsLeft = 8
	}
	bitsLeft.low -= b
	inputBits |= (inputByte >> bitsLeft)
}

20B57 = CopyUncompressedBlock(bc: blockSize) {
	{0000} = 0xA
	blockSize += 256
	de = source
	hl = destination
	repeat (blockSize or 65536) {
		{hl++} = {de++}
		if (de = 0xC000) {
			de = 0xA000
			a = sourceBank
			{0400} = ++a
			{0800} = ++a
			sourceBank = a
		}
	}
	source = de
	{0000} = 0
}

20B9D = UpdateProgress(bc: blockSize) {
	loadedBytesOfKB += blockSize
	de = laterDestination
	SVBK = laterDestinationBank
	hl = destination
	repeat (blockSize) {
		{de++} = {hl++}
		if (0xE000 <= de <= 0xE0FF) {
			SVBK = 1
			laterDestinationCallback(laterDestinationBank)
			SVBK = laterDestinationBank
			de = 0xD000
		}
	}
	SVBK = 1
	laterDestination = de
	b = 0
	while (loadedBytesOfKB >= 1024) {
		loadedBytesOfKB -= 1024
		loadedKB++
		b++
	}
	if (b != 0) {
		h, l = APIByteDivide(loadedKB, 10)
		progressString[12] = '0' + l
		progressString[13] = '0' + h
		{C1C2} = {DBF7} + 12
		APIDrawString(&progressString[12], 0, 12)
	}
}

20C33 = NextByte() a: u8 {
	{0000} = 0xA
	hl = source
	b = {hl++}
	if (hl = 0xC000) {
		hl = 0xA000
		a = sourceBank
		{0400} = ++a
		{0800} = ++a
		sourceBank = a
	}
	source = hl
	{0000} = 0
	return b
}
